# Dice Visual Polish Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Apply three visual upgrades to the combat dice cards (held-lift + HELD badge, combo accent underline bar, tray radial glow) so the runtime combat matches the April 2026 design mockup.

**Architecture:** Extend the existing `CombatDice` (scripts/combat/combat_dice.gd) with two overlay children (held badge, accent bar) and a lift tween. Add a new `DiceTrayGlow` control drawn as stepped concentric ellipses, inserted as a sibling behind `_dice_container` in `CombatScreen`, updated whenever combo accents are refreshed.

**Tech Stack:** Godot 4.x (GDScript). No new dependencies. Test runner (if used) is the existing `tests/` suite.

**Spec:** `docs/superpowers/specs/2026-04-21-dice-visual-polish-design.md`

## Task 3: DiceTrayGlow Control (new file)

**Files:**
- Create: `scripts/ui/dice_tray_glow.gd`

**Approach:** Small `Control` that draws 7 nested semi-transparent ellipses in `_draw()`, shrinking radius and alpha step-wise. Chunky pixel-art stepped gradient instead of a smooth one. Exposes `set_glow_color(Color)` and redraws on resize.

- [ ] **Step 1: Create the file**

Create `scripts/ui/dice_tray_glow.gd`:

```gdscript
extends Control
## Chunky pixel-art radial glow drawn under the combat dice tray.
## Renders N concentric semi-transparent ellipses to fake a stepped radial gradient.

const GLOW_STEPS := 7
const GLOW_ALPHA_CENTER := 0.2

var _glow_color: Color = Color(1, 1, 1, 1)


func _ready() -> void:
	mouse_filter = Control.MOUSE_FILTER_IGNORE


func set_glow_color(color: Color) -> void:
	_glow_color = color
	queue_redraw()


func _notification(what: int) -> void:
	if what == NOTIFICATION_RESIZED:
		queue_redraw()


func _draw() -> void:
	if size.x < 1.0 or size.y < 1.0:
		return
	var center := size * 0.5
	# Step from outer → inner so inner layers stack on top.
	for i in range(GLOW_STEPS, 0, -1):
		var t := float(i) / float(GLOW_STEPS)
		# Outer rings are larger, more transparent; inner rings are smaller, more opaque.
		var radius := Vector2(size.x * 0.5 * t, size.y * 0.5 * t)
		var alpha := GLOW_ALPHA_CENTER * (1.0 - t + 1.0 / float(GLOW_STEPS))
		var col := Color(_glow_color.r, _glow_color.g, _glow_color.b, alpha)
		_draw_filled_ellipse(center, radius, col)


func _draw_filled_ellipse(center: Vector2, radius: Vector2, color: Color) -> void:
	const SEGMENTS := 32
	var points := PackedVector2Array()
	for j in range(SEGMENTS):
		var angle := TAU * float(j) / float(SEGMENTS)
		points.append(center + Vector2(cos(angle) * radius.x, sin(angle) * radius.y))
	var colors := PackedColorArray()
	colors.resize(points.size())
	colors.fill(color)
	draw_polygon(points, colors)
```

```gdscript
const DICE_GLOW_RADIUS_SCALE := 1.3
const DICE_GLOW_FADE_DURATION := 0.2

var _dice_glow_tween: Tween


func _update_dice_glow(active: bool, color: Color) -> void:
	if _dice_glow == null or _dice_container == null:
		return
	if active:
		_dice_glow.set_glow_color(color)
		_resize_dice_glow()
	_fade_dice_glow(active)


func _resize_dice_glow() -> void:
	if _dice_glow == null or _dice_container == null:
		return
	var container_rect := _dice_container.get_rect()
	var glow_size := container_rect.size * DICE_GLOW_RADIUS_SCALE
	var glow_pos := container_rect.position - (glow_size - container_rect.size) * 0.5
	_dice_glow.position = glow_pos
	_dice_glow.size = glow_size


func _fade_dice_glow(active: bool) -> void:
	if _dice_glow_tween != null and _dice_glow_tween.is_valid():
		_dice_glow_tween.kill()
	var target_alpha := 1.0 if active else 0.0
	_dice_glow_tween = create_tween()
	_dice_glow_tween.tween_property(_dice_glow, "modulate:a", target_alpha, DICE_GLOW_FADE_DURATION)
```
